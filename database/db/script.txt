CREATE TABLE users 
( 
 id INT PRIMARY KEY AUTO_INCREMENT,  
 name VARCHAR NOT NULL,  
 email VARCHAR NOT NULL,  
 password VARCHAR NOT NULL,  
 role ENUM NOT NULL,  
); 

CREATE TABLE authors 
( 
 id INT PRIMARY KEY AUTO_INCREMENT,  
 name VARCHAR NOT NULL,  
 nationality VARCHAR NOT NULL,  
); 

CREATE TABLE books 
( 
 id INT PRIMARY KEY AUTO_INCREMENT,  
 title VARCHAR NOT NULL,  
 author INT NOT NULL,  
 year_publication INT,  
 page_numbers INT,  
 gender INT,  
); 

CREATE TABLE films 
( 
 id INT PRIMARY KEY AUTO_INCREMENT,  
 title VARCHAR NOT NULL,  
 gender INT,  
 synopsis VARCHAR,  
 the_cast VARCHAR,  
); 

CREATE TABLE series 
( 
 id INT PRIMARY KEY AUTO_INCREMENT,  
 title VARCHAR NOT NULL,  
 stream INT,  
 number_seasons INT NOT NULL,  
 gender INT,  
 synopsis VARCHAR,  
); 

CREATE TABLE reading_log 
( 
 the_user INT,  
 book INT,  
 status ENUM NOT NULL,  
 note FLOAT,  
); 

CREATE TABLE gender 
( 
 id INT PRIMARY KEY AUTO_INCREMENT,  
 name VARCHAR NOT NULL,  
); 

CREATE TABLE film_registration 
( 
 the_user INT,  
 film INT,  
 status ENUM NOT NULL,  
 note FLOAT,  
); 

CREATE TABLE streams 
( 
 id INT PRIMARY KEY AUTO_INCREMENT,  
 name VARCHAR NOT NULL,  
 description VARCHAR,  
); 

CREATE TABLE season_log 
( 
 the_user INT,  
 serie INT,  
 season INT NOT NULL,  
 status ENUM NOT NULL,  
); 

CREATE TYPE reading_status AS ENUM (
    'A_LER',
    'LENDO',
    'CONCLUIDO'
);

CREATE TYPE film_status AS ENUM (
    'A_VER',
    'CONCLUIDO'
);

CREATE TYPE series_status AS ENUM (
    'A_VER',
    'ASSISTINDO',
    'CONCLUIDO'
);

ALTER TABLE books ADD FOREIGN KEY(author) REFERENCES authors (author)
ALTER TABLE books ADD FOREIGN KEY(gender) REFERENCES gender (gender)
ALTER TABLE films ADD FOREIGN KEY(gender) REFERENCES gender (gender)
ALTER TABLE series ADD FOREIGN KEY(stream) REFERENCES streams (stream)
ALTER TABLE series ADD FOREIGN KEY(gender) REFERENCES gender (gender)
ALTER TABLE reading_log ADD FOREIGN KEY(the_user) REFERENCES users (the_user)
ALTER TABLE reading_log ADD FOREIGN KEY(book) REFERENCES books (book)
ALTER TABLE film_registration ADD FOREIGN KEY(the_user) REFERENCES users (the_user)
ALTER TABLE film_registration ADD FOREIGN KEY(film) REFERENCES films (film)
ALTER TABLE season_log ADD FOREIGN KEY(the_user) REFERENCES users (the_user)
ALTER TABLE season_log ADD FOREIGN KEY(serie) REFERENCES series (serie)
